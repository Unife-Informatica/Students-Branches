% --- 5. Cattura Metadati e Pagina del Titolo ---
\newcommand{\doctitle}{}
\newcommand{\docsubtitle}{}
\newcommand{\docauthor}{}
\newcommand{\docinstitute}{}
\newcommand{\docdate}{}

\let\origtitle\title
\renewcommand{\title}[1]{\gdef\doctitle{#1}\origtitle{#1}}

\providecommand{\subtitle}[1]{}
\renewcommand{\subtitle}[1]{\gdef\docsubtitle{#1}}

\let\origauthor\author
\renewcommand{\author}[1]{\gdef\docauthor{#1}\origauthor{#1}}

\providecommand{\institute}[1]{}
\renewcommand{\institute}[1]{\gdef\docinstitute{#1}}

\let\origdate\date
\renewcommand{\date}[1]{\gdef\docdate{#1}\origdate{#1}}

\renewcommand{\maketitle}{%
  \begin{titlepage}
    \thispagestyle{empty}
    \raggedright
    \vspace*{1.5cm}

    \ifx\docinstitute\empty\else
      {\small\bfseries\color{AccentColor}\MakeUppercase{\docinstitute}\par}
      \vspace{0.6cm}
    \fi

    {\color{AccentColor}\rule{3.2cm}{2.5pt}\par}
    \vspace{0.8cm}

    \ifx\doctitle\empty\else
      {\fontsize{26}{32}\selectfont\bfseries\color{PrimaryColor}\doctitle\par}
      \vspace{0.5cm}
    \fi

    \ifx\docsubtitle\empty\else
      {\Large\color{MutedText}\docsubtitle\par}
      \vspace{0.6cm}
    \fi

    \vspace{0.4cm}
    {\color{BorderGray}\rule{\linewidth}{0.8pt}\par}

    \vfill

    \ifx\docauthor\empty\else
      {\large\bfseries\color{PrimaryColor}\renewcommand{\and}{\\}\docauthor\par}
      \vspace{0.35cm}
    \fi

    \ifx\docdate\empty\else
      {\normalsize\color{MutedText}\docdate\par}
    \fi

    \vspace*{1cm}
  \end{titlepage}
}

% --- 9. Ambiente Callout (basato su TikZ, supporta paragrafi e liste) ---
\newsavebox{\calloutbox}
\newenvironment{callout}[1][Nota]{%
  \def\callouttitle{#1}%
  \begin{lrbox}{\calloutbox}%
  \begin{minipage}{\dimexpr\linewidth-24pt\relax}%
  {\small\bfseries\color{PrimaryColor}\callouttitle}\par\vspace{0.35em}\small
}{%
  \end{minipage}%
  \end{lrbox}%
  \par\vspace{0.6em}\noindent
  \begin{tikzpicture}
    \node[
      rectangle,
      rounded corners=3pt,
      draw=BorderGray,
      fill=LightBg,
      line width=0.6pt,
      inner sep=10pt
    ] (box) {\usebox{\calloutbox}};
    \fill[AccentColor] (box.north west) rectangle ([xshift=3.5pt]box.south west);
  \end{tikzpicture}
  \par\vspace{0.6em}
}
